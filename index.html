CN Lab Quick Guide

C • RPC • Ubuntu

Your commands and C programs in one place. Pick an experiment, copy a command, and follow the steps.

Important: open two terminal windows. Start the server in Terminal 1, then run the client in Terminal 2. These commands are for Ubuntu/Linux.

One-time setup

Run this once if the RPC tools are not installed.

Start here

1. Install required tools

Terminal
sudo apt update
sudo apt install rpcbind rpcsvc-proto libtirpc-dev
sudo systemctl start rpcbind

2. Create the folders

Terminal
mkdir -p ~/cn_rpc/echo ~/cn_rpc/day ~/cn_rpc/add
cd ~/cn_rpc

Folder layout: cn_rpc/echo, cn_rpc/day, and cn_rpc/add.

Experiment 7 — Echo Service Using RPC

The server returns the message sent by the client.

echo

1. Create the files

Terminal
cd ~/cn_rpc/echo
nano echo.x
nano echo_server.c
nano echo_client.c

Paste each corresponding program into its file. Save in nano with Ctrl+O, Enter, then Ctrl+X.

2. Generate RPC support files

Terminal
rpcgen -C echo.x

This generates echo.h, echo_clnt.c, echo_svc.c, and echo_xdr.c.

3. Compile

Server
gcc -I/usr/include/tirpc echo_server.c echo_svc.c echo_xdr.c -ltirpc -o echo_server
Client
gcc -I/usr/include/tirpc echo_client.c echo_clnt.c echo_xdr.c -ltirpc -o echo_client

4. Run in two terminals

Terminal 1 · Server
cd ~/cn_rpc/echo
./echo_server
Terminal 2 · Client
cd ~/cn_rpc/echo
./echo_client

5. Program files

echo.x
program ECHO_PROG {
    version ECHO_VERS {
        string ECHO(string) = 1;
    } = 1;
} = 0x23451111;
echo_server.c
#include "echo.h"

char **echo_1_svc(char **msg, struct svc_req *r)
{
    static char *result;
    result = *msg;
    return &result;
}
echo_client.c
#include <stdio.h>
#include "echo.h"

int main(void)
{
    CLIENT *c;
    char *msg = "Hello";
    char **res;

    c = clnt_create("localhost", ECHO_PROG, ECHO_VERS, "udp");
    if (c == NULL) {
        clnt_pcreateerror("localhost");
        return 1;
    }

    res = echo_1(&msg, c);
    if (res != NULL)
        printf("Echo: %s\n", *res);
    else
        clnt_perror(c, "Echo request failed");

    clnt_destroy(c);
    return 0;
}

5. Program files

day.x
program DAY_PROG {
    version DAY_VERS {
        string DAY(void) = 1;
    } = 1;
} = 0x23451112;
day_server.c
#include <time.h>
#include <string.h>
#include "day.h"

char **day_1_svc(void *arg, struct svc_req *r)
{
    static char *result;
    static char buffer[50];
    time_t now = time(NULL);

    strncpy(buffer, ctime(&now), sizeof(buffer) - 1);
    buffer[sizeof(buffer) - 1] = '\0';
    result = buffer;
    return &result;
}
day_client.c
#include <stdio.h>
#include "day.h"

int main(void)
{
    CLIENT *c;
    char **res;

    c = clnt_create("localhost", DAY_PROG, DAY_VERS, "udp");
    if (c == NULL) {
        clnt_pcreateerror("localhost");
        return 1;
    }

    res = day_1(NULL, c);
    if (res != NULL)
        printf("Date and Time: %s", *res);
    else
        clnt_perror(c, "Date/time request failed");

    clnt_destroy(c);
    return 0;
}

Experiment 9 — Addition of Two Numbers Using RPC

The client sends two integers and displays the server's sum.

add

1. Create the files

Terminal
cd ~/cn_rpc/add
nano add.x
nano add_server.c
nano add_client.c

Paste each corresponding program into its file. Save with Ctrl+O, Enter, then Ctrl+X.

2. Generate RPC support files

Terminal
rpcgen -C add.x

3. Compile

Server
gcc -I/usr/include/tirpc add_server.c add_svc.c add_xdr.c -ltirpc -o add_server
Client
gcc -I/usr/include/tirpc add_client.c add_clnt.c add_xdr.c -ltirpc -o add_client

4. Run in two terminals

Terminal 1 · Server
cd ~/cn_rpc/add
./add_server
Terminal 2 · Client
cd ~/cn_rpc/add
./add_client

5. Program files

add.x
struct numbers {
    int a;
    int b;
};

program ADD_PROG {
    version ADD_VERS {
        int ADD(numbers) = 1;
    } = 1;
} = 0x23451113;
add_server.c
#include "add.h"

int *add_1_svc(numbers *n, struct svc_req *r)
{
    static int sum;
    sum = n->a + n->b;
    return &sum;
}
add_client.c
#include <stdio.h>
#include "add.h"

int main(void)
{
    CLIENT *c;
    numbers n;
    int *res;

    printf("Enter two numbers: ");
    if (scanf("%d %d", &n.a, &n.b) != 2) {
        printf("Invalid input\n");
        return 1;
    }

    c = clnt_create("localhost", ADD_PROG, ADD_VERS, "udp");
    if (c == NULL) {
        clnt_pcreateerror("localhost");
        return 1;
    }

    res = add_1(&n, c);
    if (res != NULL)
        printf("Sum = %d\n", *res);
    else
        clnt_perror(c, "Addition request failed");

    clnt_destroy(c);
    return 0;
}
